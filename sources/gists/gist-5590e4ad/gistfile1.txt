<script>
    const canvas = document.getElementById('gameCanvas'), ctx = canvas.getContext('2d');
    canvas.width = 480; canvas.height = 420;

    let running = false, mode = "FIB", bossName = "One Million", bHP = 6, mHP = 6, gSpd = 1, hits = 0, clones = [];
    let trailData = [], stickyTime = 0, stickyLevel = 0, blastCooldown = 300, isBlasting = false;

    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    function playHitSound(isMain) {
        if(!document.getElementById('set-sound').checked || !isMain) return;
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'square';
        osc.frequency.setValueAtTime(150 + (Math.random() * 200), audioCtx.currentTime);
        gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + 0.1);
        osc.connect(gain); gain.connect(audioCtx.destination);
        osc.start(); osc.stop(audioCtx.currentTime + 0.1);
    }

    const heroes = [
        { id: "FIB", n: "Fibonacci", i: "🌀", c: "#FFD700", p: "fib", cat: "Weak" },
        { id: "FACT", n: "Factorial", i: "❗", c: "#FF4444", p: "fact", cat: "Normal" },
        { id: "POW", n: "Power", i: "🔥", c: "#FFA500", p: "pow", cat: "Normal" },
        { id: "SQR", n: "Square", i: "⬛", c: "#00ffff", p: "sqr", cat: "Normal" },
        { id: "GROW", n: "Growernacci", i: "📈", c: "#4169E1", p: "grow", cat: "Strong" },
        { id: "MULT", n: "Multiplier", i: "✖", c: "#ff00ff", p: "mult", cat: "Strong" },
        { id: "PRIM", n: "Prime", i: "#", c: "#32CD32", p: "prim", cat: "Strong" },
        { id: "BLAST", n: "Blastynacci", i: "☄️", c: "#00BFFF", p: "blast", cat: "Strong" },
        { id: "SPEEDY", n: "Speedypowernacci", i: "⚡", c: "#FF69B4", p: "speedy", cat: "Strong" },
        { id: "PENT", n: "Pentanation", i: "🌌", c: "#800080", p: "pent", cat: "Strong" },
        { id: "BB", n: "Busy Beaver", i: "🦫", c: "#8B4513", p: "bb", cat: "Superman level" },
        { id: "OMEGA", n: "Omega Beaver", i: "🧬", c: "#4B0082", p: "omega", cat: "Superman level" }
    ];

    let bosses = [
        { n: "One Million", l: 6, cat: "Weak" }, 
        { n: "One Trillion", l: 12, cat: "Weak" }, 
        { n: "Avogadro", l: 23, cat: "Normal" }, 
        { n: "Eddington", l: 79, cat: "Normal" },
        { n: "Googol", l: 100, cat: "Normal" }, 
        { n: "Googolplex", l: 100000000, cat: "Strong" }, 
        { n: "Skewes' Number", l: 340, cat: "Strong" },
        { n: "Grandma's Cookies", l: 500, cat: "Strong" }, 
        { n: "Graham's Number", l: 1000000, cat: "Strong" }, 
        { n: "Tree(3)", l: 50000000, cat: "Strong" },
        { n: "SCG(13)", l: 99999999, cat: "Absolute titan" }, 
        { n: "Rayo's Number", l: 500000000, cat: "Absolute titan" }, 
        { n: "Absolute Infinity", l: 9999999999, cat: "Absolute titan" }
    ];

    function toggleDarkMode() { document.body.classList.toggle('dark-ui', document.getElementById('set-dark').checked); }
    function openMenu(id) { document.querySelectorAll('.overlay-menu').forEach(m => m.style.display = 'none'); if(id !== 'none') document.getElementById(id).style.display = 'flex'; }
    function backToMain() { openMenu('starter-menu'); }
    function setSpd(s) { gSpd = s; document.querySelectorAll('.spd-btn').forEach(b => b.classList.remove('active-spd')); document.getElementById('btn-'+s).classList.add('active-spd'); }

    function summonCustomBoss() {
        let n = document.getElementById('custom-boss-name').value || "Titan";
        let h = document.getElementById('custom-boss-hp').value;
        let l = h.includes('^') ? parseFloat(h.split('^')[1]) * Math.log10(parseFloat(h.split('^')[0])) : parseFloat(h);
        if(!isNaN(l)) { bosses.push({n:n, l:l, cat: "Absolute titan"}); renderMenus(); }
    }

    function renderMenus() {
        const hL = document.getElementById('hero-list'); hL.innerHTML = "";
        heroes.forEach(h => {
            let d = document.createElement('div'); d.className = `list-item ${mode === h.id ? 'selected' : ''}`;
            d.innerHTML = `${h.i} ${h.n} [${h.cat}]`; d.onclick = () => { mode = h.id; renderMenus(); };
            hL.appendChild(d);
        });
        const bL = document.getElementById('boss-list'); bL.innerHTML = "";
        bosses.forEach(b => {
            let d = document.createElement('div'); d.className = `list-item ${bossName === b.n ? 'selected' : ''}`;
            d.innerHTML = `👹 ${b.n} (10^${b.l > 1000 ? b.l.toExponential(1) : b.l.toFixed(0)}) [${b.cat}]`; 
            d.onclick = () => { bossName = b.n; bHP = b.l; mHP = b.l; renderMenus(); };
            bL.appendChild(d);
        });
    }

    let bObj = { x:240, y:210, r:55, vx:2.2, vy:2.2 };
    let hObj = { x:80, y:80, r:20, vx:4.5, vy:4.5 };

    function onHit(o) {
        hits++; let hL = 0;
        let hData = heroes.find(h => h.id === mode);
        let m = 1;
        
        playHitSound(o === hObj); 

        if(hData.p === "blast") m = 1.3;
        if(hData.p === "speedy") { m = 1.5; hObj.vx *= 1.01; hObj.vy *= 1.01; }

        if(hData.p === "bb") hL = Math.pow(1.8, hits);
        else hL = (Math.pow(hits, 1.1) + 2) * m;

        if(hData.p === "omega" && clones.length < 30) clones.push({x:o.x, y:o.y, r:12, vx:Math.random()*10-5, vy:Math.random()*10-5, color: hData.c});

        bHP -= (hL / mHP) * 7;
        document.getElementById('hp-fill').style.width = Math.max(0, (bHP/mHP*100)) + "%";
        document.getElementById('boss-label').innerText = `10^${Math.max(0, bHP).toFixed(1)}`;
        document.getElementById('dmg-tracker').innerText = "POWER: 10^" + hL.toFixed(2);
    }

    function solidMove(o) {
        if(mode === "BLAST" && isBlasting && o !== bObj) {
            o.x += (bObj.x - o.x) * 0.15; o.y += (bObj.y - o.y) * 0.15;
            return;
        }

        let steps = Math.ceil(4 * gSpd);
        for(let i=0; i<steps; i++){
            o.x += (o.vx * gSpd)/steps; o.y += (o.vy * gSpd)/steps;
            
            if(o.x < o.r) { o.x = o.r; o.vx *= -1; }
            if(o.x > 480 - o.r) { o.x = 480 - o.r; o.vx *= -1; }
            if(o.y < o.r) { o.y = o.r; o.vy *= -1; }
            if(o.y > 420 - o.r) { o.y = 420 - o.r; o.vy *= -1; }

            if(o !== bObj){
                let dx = o.x - bObj.x, dy = o.y - bObj.y;
                let dist = Math.sqrt(dx*dx + dy*dy);
                if(dist < o.r + bObj.r){
                    let ang = Math.atan2(dy, dx);
                    o.x = bObj.x + Math.cos(ang) * (o.r + bObj.r + 1);
                    o.y = bObj.y + Math.sin(ang) * (o.r + bObj.r + 1);
                    o.vx = Math.cos(ang) * 5; o.vy = Math.sin(ang) * 5;
                    onHit(o); break;
                }
            }
        }
        if(document.getElementById('set-trails').checked) trailData.push({x:o.x, y:o.y, r:o.r, a:0.3});
    }

    function update() {
        if (!running) return;
        ctx.fillStyle = document.getElementById('set-dark').checked ? "#000" : "#fff";
        ctx.fillRect(0,0,480,420);

        if(mode === "BLAST") {
            blastCooldown -= gSpd;
            if(blastCooldown <= 0) {
                isBlasting = true;
                ctx.beginPath(); ctx.strokeStyle = "#00BFFF"; ctx.lineWidth = 12;
                ctx.moveTo(hObj.x, hObj.y); ctx.lineTo(bObj.x, bObj.y); ctx.stroke();
                onHit(hObj);
                if(blastCooldown < -50) { blastCooldown = 300; isBlasting = false; }
            }
        }

        trailData.forEach((t, i) => {
            ctx.beginPath(); ctx.arc(t.x, t.y, t.r, 0, 7);
            ctx.fillStyle = `rgba(180,180,180,${t.a})`; ctx.fill();
            t.a -= 0.02; if(t.a <= 0) trailData.splice(i, 1);
        });

        if(document.getElementById('set-glow').checked) ctx.shadowBlur = 10; ctx.shadowColor = "red";
        solidMove(bObj); solidMove(hObj); clones.forEach(c => solidMove(c));
        ctx.shadowBlur = 0;

        ctx.fillStyle="#333"; ctx.beginPath(); ctx.arc(bObj.x, bObj.y, bObj.r, 0, 7); ctx.fill();
        ctx.fillStyle="#fff"; ctx.textAlign="center"; ctx.fillText("TITAN", bObj.x, bObj.y+5);

        let hData = heroes.find(h => h.id === mode);
        [hObj, ...clones].forEach((h, i) => {
            ctx.fillStyle = (i===0) ? hData.c : h.color;
            ctx.beginPath(); ctx.arc(h.x, h.y, h.r, 0, 7); ctx.fill();
            if(i===0){ ctx.fillStyle="#fff"; ctx.fillText(hData.i, h.x, h.y+5); }
        });

        if(bHP <= 0) { running = false; openMenu('win-screen'); }
        else requestAnimationFrame(update);
    }

    function startGame() { 
        if(audioCtx.state === 'suspended') audioCtx.resume();
        running=true; trailData=[]; clones=[]; bHP=mHP; hits=0; stickyTime=0; blastCooldown=300;
        hObj.vx = 4.5; hObj.vy = 4.5;
        openMenu('none'); document.getElementById('top-bar').style.display='flex'; document.getElementById('exit-btn').style.display='block'; update(); 
    }
    function exitToMenu() { running=false; backToMain(); document.getElementById('top-bar').style.display='none'; document.getElementById('exit-btn').style.display='none'; }
    renderMenus();
</script>
